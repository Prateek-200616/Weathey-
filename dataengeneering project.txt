// --- Helper Functions (Simplified as API logic is removed) ---

// This function is now only used to map the initial cities to mock data
const getMockDataForCity = (cityName) => {
    // Find mock data for the city
    const data = MOCK_WEATHER_DATA.find(d => d.cityName.toLowerCase() === cityName.toLowerCase());
    
    if (data) {
        // Return a fresh copy with updated timestamp and randomized temp
        const tempF = ((parseFloat(data.temperatureC) + (Math.random() * 0.5 - 0.25)) * 9/5 + 32).toFixed(1);

        return { 
            ...data, 
            timestamp: new Date().toLocaleTimeString(),
            // Simulate realistic variation on refresh
            temperatureC: (parseFloat(data.temperatureC) + (Math.random() * 0.5 - 0.25)).toFixed(1),
            temperatureF: tempF,
        };
    }
    
    // Fallback: If city not found, use the generic mock data instead of returning an error object
    const genericData = MOCK_WEATHER_DATA.find(d => d.cityName === '__GENERIC__');
    
    if (genericData) {
        return {
            ...genericData,
            cityName: cityName, // Use the requested city name
            description: `Simulated: ${genericData.description}`,
            timestamp: new Date().toLocaleTimeString(),
        };
    }

    // Final fallback to error object if even generic data is missing
    return { 
        cityName: cityName, 
        error: true, 
        country: 'N/A', 
        temperatureC: 'N/A', 
        temperatureF: 'N/A',
        condition: 'Unknown',
        description: 'No mock data available',
        humidity: 'N/A',
        pressure: 'N/A',
        windSpeed: 'N/A',
        sunrise: 'N/A',
        sunset: 'N/A',
        timestamp: new Date().toLocaleTimeString(),
    };
};

// --- Custom Components ---

  // If we couldn't find mock data, show failure state
  if (!data || data.error) return (
    <div className="bg-white p-6 rounded-xl shadow-lg border border-gray-100 transition duration-300 transform hover:scale-[1.02] col-span-1 min-h-[350px] flex items-center justify-center">
        <p className="text-gray-500 font-semibold">Failed to load simulated weather for {data?.cityName}.</p>
    </div>
  );

  const IconMap = {
    'Clouds': <Cloud className="w-16 h-16 text-indigo-500" />,
    'Clear': <Sunrise className="w-16 h-16 text-yellow-500" />,

  return (

      {/* City and Condition */}

// --- Main Application Component ---

const App = () => {
  // Use a smaller initial list to ensure mock data is displayed correctly
  const initialCities = ['London', 'New York', 'Tokyo', 'Paris', 'Sydney', 'Delhi', 'Mumbai']; 
  const [weatherData, setWeatherData] = useState([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);
  const [newCity, setNewCity] = useState('');
  const [cities, setCities] = useState(initialCities);

  // SIMULATED fetch function
  const fetchWeatherForCity = useCallback(async (city) => {
    // Simulate network delay
    await new Promise(resolve => setTimeout(resolve, 300));
    
    const mockData = getMockDataForCity(city);
    
    // Since getMockDataForCity now returns generic data instead of an error object, 
    // we only log an error if the final fallback was hit (which should be rare).
    if (mockData.error) {
        console.error(`Final fallback: No mock data available for ${city}.`);
        return mockData;
    }

    // We only log if it used the generic fallback for better debugging
    if (mockData.description.includes("Simulated:")) {
        console.warn(`City '${city}' not found in specific mock list. Using generic mock data.`);
    }

    return mockData;
  }, []);

  // SIMULATED fetch all function
  const fetchAllWeather = useCallback(async () => {
    setLoading(true);
    setError(null);
    
    // For every city in the 'cities' array, get the corresponding mock data
    try {
        const results = await Promise.all(cities.map(city => fetchWeatherForCity(city)));
        setWeatherData(results.filter(data => data !== null)); 
    } catch (err) {
      setError("Failed to simulate weather data refresh.");
    } finally {
      setLoading(false);
    }
  }, [cities, fetchWeatherForCity]);

  // Initial load
  useEffect(() => {
    fetchAllWeather();
  }, [fetchAllWeather]);

  const handleAddCity = () => {
    const city = newCity.trim();
    if (city && !cities.map(c => c.toLowerCase()).includes(city.toLowerCase())) {
      setCities([...cities, city]);
      setNewCity('');
    }
  };

  // Re-fetch data whenever the city list changes
  useEffect(() => {
      // Re-run fetchAllWeather to include the new city
      fetchAllWeather();
  }, [cities, fetchAllWeather]);


  return (
    <div className="min-h-screen bg-gray-50 p-4 sm:p-8 font-sans">
      <header className="mb-8 p-4 bg-white shadow-md rounded-lg">
        <h1 className="text-4xl font-extrabold text-indigo-700 text-center tracking-tight">
          Simulated Weather Dashboard
        </h1>
        <p className="text-center text-gray-500 mt-2">
            Displaying mock data. Real API key is no longer required.
        </p>
      </header>
      
      {/* City Input and Control */}
      <div className="max-w-xl mx-auto mb-8 flex space-x-2">
        <input
          type="text"
          value={newCity}
          onChange={(e) => setNewCity(e.target.value)}
          placeholder="Enter new city (e.g., Shanghai)"
          className="flex-grow p-3 border border-gray-300 rounded-lg focus:ring-indigo-500 focus:border-indigo-500 transition duration-150"
        />
        <button
          onClick={handleAddCity}
          className="px-4 py-3 bg-indigo-600 text-white font-semibold rounded-lg shadow-md hover:bg-indigo-700 transition duration-150 transform hover:scale-[1.05]"
        >
          Add City
        </button>
        <button
          onClick={fetchAllWeather}
          disabled={loading}
          className="px-4 py-3 bg-green-600 text-white font-semibold rounded-lg shadow-md hover:bg-green-700 transition duration-150 transform hover:scale-[1.05] disabled:opacity-50"
        >
          {loading ? 'Simulating...' : 'Refresh Mock Data'}
        </button>
      </div>

      {/* Error/Loading Status */}
      {(loading || error) && (
        <div className={`p-4 mb-6 rounded-lg max-w-4xl mx-auto text-center ${error ? 'bg-red-100 text-red-700' : 'bg-blue-100 text-blue-700'}`}>
          {error || "Simulating data retrieval..."}
        </div>
      )}

      {/* Weather Grid */}
      <div className="max-w-6xl mx-auto grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
        {weatherData.map((data, index) => (
          <WeatherCard key={index} data={data} />
        ))}
      </div>
      
      {/* API Key Warning removed */}
    </div>
  );
};

export default App;
